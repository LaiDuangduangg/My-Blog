## Description

如题，已知一个数列 `a`，你需要进行下面三种操作：

- 将某区间每一个数乘上 `x`；
- 将某区间每一个数加上 `x`；
- 求出某区间每一个数的和。

## Input

第一行包含三个整数 `n`、`q`、`m`，分别表示该数列数字的个数、操作的总个数和模数。

第二行包含 `n` 个用空格分隔的整数，其中第 `i` 个数字表示数列第 `i` 项的初始值 `a_i`。

接下来 `q` 行每行包含若干个整数，表示一个操作，具体如下：

操作 1：

```text
1 x y k
```

含义：将区间 `[x, y]` 内每个数乘上 `k`。

操作 2：

```text
2 x y k
```

含义：将区间 `[x, y]` 内每个数加上 `k`。

操作 3：

```text
3 x y
```

含义：输出区间 `[x, y]` 内每个数的和对 `m` 取模所得的结果。

## Output

输出包含若干行整数，即为所有操作 3 的结果。

## Sample 1

### Input

```text
5 5 38
1 5 4 2 3
2 1 4 1
3 2 5
1 2 4 2
2 3 5 5
3 1 4
```

### Output

```text
17
2
```

## Hint

### 数据范围

对于 `30%` 的数据：`n ≤ 8`，`q ≤ 10`。

对于 `70%` 的数据：`n ≤ 10^3`，`q ≤ 10^4`。

对于 `100%` 的数据：`1 ≤ n ≤ 10^5`，`1 ≤ q ≤ 10^5`，`1 ≤ a_i, k ≤ 10^4`。

除样例外，`m = 571373`。

## 代码如下:
```cpp
#include <bits/stdc++.h>
using namespace std;

long long n, m, q;
constexpr int MAXN = 1e5 + 10;

vector <long long> arr (MAXN);
vector <long long> sum (MAXN << 2); // 区间和数组
vector <long long> add (MAXN << 2); // 懒标记add数组,  + //区间内每个数加多少
vector <long long> mul (MAXN << 2, 1); // 懒标记mul数组,  * // 区间内每个数乘多少

void mul_lazy (int i, long long jobv) {
    sum[i] = sum[i] * jobv % m;
    add[i] = add[i] * jobv % m;
    mul[i] = mul[i] * jobv % m;
}

void add_lazy (int i, long long jobv, int n) {
    sum[i] = (sum[i] + jobv * n) % m;
    add[i] = (add[i] + jobv) % m;
}

// 将孩子的值上传回父亲
void PushUp (int i) {
    sum[i] = (sum[i << 1] + sum[i << 1 | 1]) % m;
}

void PushDown (long long i, int lenl, int lenr) {
    if (mul[i] != 1) {
        mul_lazy (i << 1, mul[i]);
        mul_lazy (i << 1 | 1, mul[i]);
        mul[i] = 1;
    }
    if (add[i] != 0) {
        add_lazy (i << 1, add[i], lenl);
        add_lazy (i << 1 | 1, add[i], lenr);
        add[i] = 0;
    }
}

void build (int l, int r, int i) {
    if (l == r) {
        sum[i] = arr[l] % m;
        return;
    }
    else {
        long long mid = (l + r) >> 1;
        build (l, mid, i << 1);
        build (mid + 1, r, i << 1 | 1);

        PushUp (i);
    }
}

void range_multi (int jobl, int jobr, int jobv, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        mul_lazy (i, jobv);
        return;
    }
    else {
        long long mid = (l + r) >> 1;
        PushDown(i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            range_multi (jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            range_multi (jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        PushUp (i);
    }
}

void range_add (int jobl, int jobr, int jobv, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        add_lazy (i, jobv, r - l + 1);
        return;
    }
    else {
        long long mid = (l + r) >> 1;
        PushDown (i, mid - l + 1, r - mid);
        if (jobl <= mid) {
            range_add (jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            range_add (jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        PushUp (i);
    }
}

long long query (int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        return sum[i];
    }
    
    long long ans = 0;
    long long mid = (l + r) >> 1;
    PushDown (i, mid - l + 1, r - mid);
    if (jobl <= mid) {
        ans = (ans + query (jobl, jobr, l, mid, i << 1)) % m;
    }
    if (jobr > mid) {
        ans = (ans + query (jobl, jobr, mid + 1, r, i << 1 | 1)) % m;
    }
    ans = ans % m;
    return ans;
}

int main ()
{
    cin >> n >> q >> m;

    for (int i = 1; i <= n; i ++)  cin >> arr[i];
    build (1, n, 1);

    while (q --) {
        int op;
        cin >> op;

        if (op == 1) {
            long long l, r, val;
            cin >> l >> r >> val;

            range_multi (l, r, val, 1, n, 1);
        }
        if (op == 2) {
            long long l, r, val;
            cin >> l >> r >> val;

            range_add (l, r, val, 1, n, 1);
        }
        if (op == 3) {
            long long l, r;
            cin >> l >> r;

            long long ans = query (l, r, 1, n, 1);
            cout << ans << '\n';
        }
    }
    return 0;
}
```
