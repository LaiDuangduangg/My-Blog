# 本题是关于区间重置与区间增加之间两个懒标记之间的优先级处理线段树



## 题目描述

给定一个长度为 $n$ 的序列 $a$，要求支持如下三种操作：

1. 给定区间 $[l,r]$，将区间内每个数都修改为 $x$。
2. 给定区间 $[l,r]$，将区间内每个数都加上 $x$。
3. 给定区间 $[l,r]$，求区间内的最大值。

## 输入格式

第一行是两个整数，依次表示序列的长度 $n$ 和操作的个数 $q$。

第二行有 $n$ 个整数，第 $i$ 个整数表示序列中的第 $i$ 个数 $a_i$。

接下来 $q$ 行，每行表示一个操作。每行首先有一个整数 $op$，表示操作的类型。

- 若 $op=1$，则接下来有三个整数 $l,r,x$，表示将区间 $[l,r]$ 内的每个数都修改为 $x$。
- 若 $op=2$，则接下来有三个整数 $l,r,x$，表示将区间 $[l,r]$ 内的每个数都加上 $x$。
- 若 $op=3$，则接下来有两个整数 $l,r$，表示查询区间 $[l,r]$ 内的最大值。

## 输出格式

对于每个 $op=3$ 的操作，输出一行一个整数表示答案。

## 样例 1

### 输入

```text
6 6
1 1 4 5 1 4
1 1 2 6
2 3 4 2
3 1 4
3 2 3
1 1 6 -1
3 1 6
```

### 输出

```text
7
6
-1
```

## 样例 2

### 输入

```text
4 4
10 4 -3 -7
1 1 3 0
2 3 4 -4
1 2 4 -9
3 1 4
```

### 输出

```text
0
```

## 数据规模与约定

- 对于 $10\%$ 的数据，$n=q=1$。
- 对于 $40\%$ 的数据，$n,q\le 10^3$。
- 对于 $50\%$ 的数据，$0\le a_i,x\le 10^4$。
- 对于 $60\%$ 的数据，$op\ne 1$。
- 对于 $90\%$ 的数据，$n,q\le 10^5$。
- 对于 $100\%$ 的数据，$1\le n,q\le 10^6$，$1\le l,r\le n$，$op\in\{1,2,3\}$，$|a_i|,|x|\le 10^9$。

## 提示

请注意大量数据读入对程序效率造成的影响。


## 代码如下:
```cpp
#include <bits/stdc++.h>
using namespace std;

constexpr int MAXN = 1e6 + 10;

vector <long long> arr;
vector <long long> add;
vector <bool> if_update;
vector <long long> change;
vector <long long> Max;

void update_lazy (int i, int jobv) {
    Max[i] = jobv; // 因为全部被重置，所以最大值直接被重置成jobv
    add[i] = 0; // add值直接被置零,之前的add全部无效掉
    change[i] = jobv;
    if_update[i] = true;
}

void add_lazy (long long i, long long jobv) {
    add[i] += jobv;
    Max[i] += jobv;
}

void PushUp (long long i) {
    Max[i] = max (Max[i << 1], Max[i << 1 | 1]);
}

void PushDown(long long i) {
    if (if_update[i]) {
        update_lazy (i << 1, change[i]);
        update_lazy (i << 1 | 1, change[i]);   
        if_update[i] = false;
    }
    if (add[i]) {
        add_lazy (i << 1, add[i]);
        add_lazy (i << 1 | 1, add[i]);
        add[i] = 0;
    }
}

// 更新任务
void update(long long jobl, long long jobr, long long jobv, long long l, long long r, long long i) {
    if (jobl <= l && r <= jobr) {
        update_lazy (i, jobv);
        return;
    }
    else {
        int mid = (l + r) >> 1;
        PushDown (i);
        if (jobl <= mid) {
            update (jobl, jobr, jobv, l, mid, i << 1);
        }
        if (jobr > mid) {
            update (jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
        }
        PushUp (i);
    }
}

void range_add(long long jobl, long long jobr, long long jobv, long long l, long long r, long long i) {
    if (jobl <= l && r <= jobr) {
        add_lazy(i, jobv);
        return;
    }
    
    PushDown (i);
    int mid = (l + r) >> 1;
    if (jobl <= mid) {
        range_add (jobl, jobr, jobv, l, mid, i << 1);
    }
    if (jobr > mid) {
        range_add (jobl, jobr, jobv, mid + 1, r, i << 1 | 1);
    }
    PushUp (i);
}

long long query(long long jobl, long long jobr, long long l, long long r, long long i) {
    if (jobl <= l && r <= jobr) {
        return Max[i];
    }

    long long mid = (l + r) >> 1;
    PushDown (i);

    long long ans = LLONG_MIN;
    if (jobl <= mid) {
        ans = max (ans, query (jobl, jobr, l, mid, i << 1));
    }
    if (jobr > mid) {
        ans = max (ans, query (jobl, jobr, mid + 1, r, i << 1 | 1));
    }

    return ans;
}

void build(long long l, long long r, long long i) {
    if (l == r) {
        Max[i] = arr[l];
        return;
    }

    long long mid = (l + r) >> 1;
    build(l, mid, i << 1);
    build(mid + 1, r, i << 1 | 1);
    PushUp(i);

    add[i] = 0;
    change[i] = 0;
    if_update[i] = false;
}

int main () 
{
    ios::sync_with_stdio (false);
    cin.tie (0);

    long long n, q;
    cin >> n >> q;

    int size = 4 * n + 5;
    arr.assign (size, 0);
    add.assign(size, 0);
    Max.assign(size, 0);
    change.assign(size, 0);
    if_update.assign(size, 0);

    for (int i = 1; i <= n; i ++) {
        cin >> arr[i];
    }    
    build (1, n, 1);
    while (q --) {
        int op;
        cin >> op;

        if (op == 1) {
            int l, r, x;
            cin >> l >> r >> x;
            update (l, r, x, 1, n, 1);
        }
        if (op == 2) {
            long long l, r, x;
            cin >> l >> r >> x;
            range_add (l, r, x, 1, n, 1);
        }
        if (op == 3) {
            long long l, r;
            cin >> l >> r;
            long long outt = query (l, r, 1, n, 1);
            cout << outt << '\n';
        }
    }
    return 0;
}
```
