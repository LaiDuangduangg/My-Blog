## 单点修改,区间查询
```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int n, q;
vector<ll> a, tr;

// 用两个儿子的答案更新父亲
void pushUp(int p) {
    tr[p] = tr[p << 1] + tr[p << 1 | 1];
}

// 建树：让 tr[p] 表示 a[l..r] 的和
void build(int p, int l, int r) {
    if (l == r) {
        tr[p] = a[l];
        return;
    }

    int mid = (l + r) >> 1;

    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);

    pushUp(p);
}

// 把 a[pos] 修改为 value
void modify(int p, int l, int r, int pos, ll value) {
    if (l == r) {
        tr[p] = value;
        return;
    }

    int mid = (l + r) >> 1;

    if (pos <= mid) {
        modify(p << 1, l, mid, pos, value);
    } else {
        modify(p << 1 | 1, mid + 1, r, pos, value);
    }

    pushUp(p);
}

// 查询 a[L..R] 的和
ll query(int p, int l, int r, int L, int R) {
    if (L <= l && r <= R) {
        return tr[p];
    }

    int mid = (l + r) >> 1;
    ll ans = 0;

    if (L <= mid) {
        ans += query(p << 1, l, mid, L, R);
    }

    if (R > mid) {
        ans += query(p << 1 | 1, mid + 1, r, L, R);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> q;

    a.assign(n + 1, 0);
    tr.assign(4 * n + 5, 0);

    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }

    build(1, 1, n);

    while (q--) {
        int op;
        cin >> op;

        if (op == 1) {
            int x;
            ll v;
            cin >> x >> v;
            modify(1, 1, n, x, v);
        } else {
            int l, r;
            cin >> l >> r;
            cout << query(1, 1, n, l, r) << '\n';
        }
    }

    return 0;
}
```

## 区间修改,区间查询
```cpp
#include <bits/stdc++.h>
using namespace std;

using ll = long long;

int n, q;
vector<ll> a, tr, lazy;

// 用左右儿子的答案更新父亲
void pushUp(int p) {
    tr[p] = tr[p << 1] + tr[p << 1 | 1];
}

// 给当前节点代表的整个区间 [l, r] 加 add
void apply(int p, int l, int r, ll add) {
    tr[p] += add * (r - l + 1);
    lazy[p] += add;
}

// 将当前节点的懒标记下传给两个儿子
void pushDown(int p, int l, int r) {
    if (lazy[p] == 0 || l == r) {
        return;
    }

    int mid = (l + r) >> 1;

    apply(p << 1, l, mid, lazy[p]);
    apply(p << 1 | 1, mid + 1, r, lazy[p]);

    lazy[p] = 0;
}

// 建树
void build(int p, int l, int r) {
    if (l == r) {
        tr[p] = a[l];
        return;
    }

    int mid = (l + r) >> 1;

    build(p << 1, l, mid);
    build(p << 1 | 1, mid + 1, r);

    pushUp(p);
}

// 给 a[L..R] 的每个元素加 add
void rangeAdd(int p, int l, int r, int L, int R, ll add) {
    if (L <= l && r <= R) {
        apply(p, l, r, add);
        return;
    }

    pushDown(p, l, r);

    int mid = (l + r) >> 1;

    if (L <= mid) {
        rangeAdd(p << 1, l, mid, L, R, add);
    }

    if (R > mid) {
        rangeAdd(p << 1 | 1, mid + 1, r, L, R, add);
    }

    pushUp(p);
}

// 查询 a[L..R] 的和
ll query(int p, int l, int r, int L, int R) {
    if (L <= l && r <= R) {
        return tr[p];
    }

    pushDown(p, l, r);

    int mid = (l + r) >> 1;
    ll ans = 0;

    if (L <= mid) {
        ans += query(p << 1, l, mid, L, R);
    }

    if (R > mid) {
        ans += query(p << 1 | 1, mid + 1, r, L, R);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n >> q;

    a.assign(n + 1, 0);
    tr.assign(4 * n + 5, 0);
    lazy.assign(4 * n + 5, 0);

    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }

    build(1, 1, n);

    while (q--) {
        int op;
        cin >> op;

        if (op == 1) {
            int l, r;
            ll d;
            cin >> l >> r >> d;

            rangeAdd(1, 1, n, l, r, d);
        } else if (op == 2) {
            int l, r;
            cin >> l >> r;

            cout << query(1, 1, n, l, r) << '\n';
        }
    }

    return 0;
}
```
