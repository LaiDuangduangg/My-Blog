## 单点修改,区间查询
```cpp
#include <bits/stdc++.h>
using namespace std;

int n, q;
vector<long long> a, tr;

// 用两个儿子的答案更新父亲
void pushUp(int i) {
    tr[i] = tr[i << 1] + tr[i << 1 | 1];
}

// 建树：让 tr[p] 表示 a[l..r] 的和
void build(int l, int r, int i) {
    if (l == r) {
        tr[i] = a[l];
    }
    else {
        int mid = (l + r) >> 1;

        build(p << 1, l, mid);
        build(p << 1 | 1, mid + 1, r);

        pushUp(p);
    }
}

// 把 a[pos] 修改为 value
void modify(int jobi, int jobv,int l, int r, int i) {
    if (l == r) {
        tr[i] = jobv;
        return;
    }
    else {
        int mid = (l + r) >> 1;

        if (pos <= mid) {
            modify(p << 1, l, mid, pos, value);
        } 
        else {
            modify(p << 1 | 1, mid + 1, r, pos, value);
        }
        pushUp(p);
    }
}

// 查询 a[L..R] 的和
ll query(int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        return tr[i];
    }
    
    int mid = (l + r) >> 1;
    long long ans = 0;

    if (jobl <= mid) {
        ans += query(jobl, jobr, l, mid, i << 1);
    }

    if (jobr > mid) {
        ans += query(jobl, jobr, mid + 1, r, i << 1 | 1);
    }
 
    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> q;

    a.assign(n + 1, 0);
    tr.assign(4 * n + 5, 0);

    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }

    build(1, n, 1);

    while (q--) {
        int op;
        cin >> op;

        if (op == 1) {
            int x;
            long long v;
            cin >> x >> v;
            modify(x, v, 1, n, 1);
        } else {
            int l, r;
            cin >> l >> r;
            cout << query(l, r, 1, n, 1) << '\n';
        }
    }

    return 0;
}
```

## 区间修改,区间查询
```cpp
#include <bits/stdc++.h>
using namespace std;

int n, q;
vector<long long> a, tr, add;

// 用左右儿子的答案更新父亲
void pushUp(int p) {
    tr[p] = tr[p << 1] + tr[p << 1 | 1];
}

// 懒信息下发
// lenl : 左侧范围有几个数
// lenr : 右侧范围有几个数
void pushDown(int i, int lenl, int lenr) {  
    if (add[i] != 0) {
        lazy (i << 1, add[i], lenl);
        lazy (i << 1 | 1, add[i], lenr);
        add[i] = 0;
    }
}

// 给当前节点代表的整个区间 [l, r] 加 val
void lazy(int i, long long val, int n) {
    tr[i] += val * n;
    add[i] += val;
}

// 建树
void build(int l, int r, int i) {
    if (l == r) {
        tr[i] = a[l];
        return;
    }

    int mid = (l + r) >> 1;

    build(i << 1, l, mid);
    build(i << 1 | 1, mid + 1, r);

    pushUp(i);
}

// 给 a[jobl..jobr] 的每个元素加 job_val
void rangeAdd(int jobl, int jobr, long long job_val, int l, int r, int i) {
    if (L <= l && r <= R) {
        lazy(p, l, r, add);
    }
    else {
        int mid = (l + r) >> 1;
        pushDown(i, mid - l + 1, r - mid);

        if (l <= mid) {
            rangeAdd(jobl, jobr, job_val, l, mid, i << 1);
        }
        if (r > mid) {
            rangeAdd(jobl, jobr, job_val, mid + 1, r, i << 1 | 1);
        }   

        // 汇总自己的信息，即汇总自己孩子的信息总和
        pushUp(i);
    }
}

// 查询 a[L..R] 的和
long long query(int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        return tr[i];
    }

    pushDown(i, mid - l + 1, r - mid);

    int mid = (l + r) >> 1;
    long long ans = 0;

    if (L <= mid) {
        ans += query(i << 1, l, mid, L, R);
    }
    if (R > mid) {
        ans += query(i << 1 | 1, mid + 1, r, L, R);
    }

    return ans;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);

    cin >> n >> q;

    a.assign(n + 1, 0);
    tr.assign(4 * n + 5, 0);
    lazy.assign(4 * n + 5, 0);

    for (int i = 1; i <= n; ++i) {
        cin >> a[i];
    }

    build(1, n, 1);

    while (q--) {
        int op;
        cin >> op;

        if (op == 1) {
            int l, r;
            ll d;
            cin >> l >> r >> d;

            rangeAdd(l, r, d, 1, n, 1);
        } else if (op == 2) {
            int l, r;
            cin >> l >> r;

            cout << query(l, r, 1, n, 1) << '\n';
        }
    }

    return 0;
}
```
