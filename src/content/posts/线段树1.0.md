## Background

很多学校流行一种比较的习惯。老师们很喜欢询问，从某某到某某当中，分数最高的是多少。这让很多学生很反感。

## Description

不管你喜不喜欢，现在需要你做的，就是按照老师的要求，写一个程序，模拟老师的询问。当然，老师有时候需要更新某位同学的成绩。

## Input

第一行，有两个正整数 `n` 和 `m`（`0 < n ≤ 2 × 10^5`，`0 < m ≤ 2 × 10^5`），分别代表学生的数目和操作的数目。学生 ID 编号分别从 `1` 编到 `n`。

第二行包含 `n` 个整数，代表这 `n` 个学生的初始成绩，其中第 `i` 个数代表 ID 为 `i` 的学生的成绩，保证学生的成绩为 `1 ∼ 10^9` 之间的正整数。

接下来有 `m` 行。每一行有一个字符 `c`（只取 `Q` 或 `U`），和两个正整数 `a`、`b`。

- 当 `c` 为 `Q` 的时候，表示这是一条询问操作，它询问 ID 从 `a` 到 `b`（包括 `a`、`b`）的学生当中，成绩最高的是多少；
- 当 `c` 为 `U` 的时候，表示这是一条更新操作，如果当前 `a` 学生的成绩低于 `b`，则把 ID 为 `a` 的学生的成绩更改为 `b`，否则不改动。

## Output

对于每一次询问操作输出一行一个整数，表示最高成绩。

## Sample 1

### Input

```text
5 6
1 2 3 4 5
Q 1 5
U 3 6
Q 3 4
Q 4 5
U 2 9
Q 1 5
```

### Output

```text
5
6
5
9
```

## 代码如下:
```cpp
#include <bits/stdc++.h>
using namespace std;

vector <long long> arr;
vector <long long> Max;

void PushUp (long long i) {
    Max[i] = max (Max[i << 1], Max[i << 1 | 1]);
}

void build (int l, int r, int i) {
    if (l == r) {
        Max[i] = arr[l];
        return;
    }
    else {
        int mid = (l + r) >> 1;
        build (l, mid, i << 1);
        build (mid + 1, r, i << 1 | 1);

        PushUp (i);
    }
}

long long query (int jobl, int jobr, int l, int r, int i) {
    if (jobl <= l && r <= jobr) {
        return Max[i];
    }
    // 没有PushDown
    long long ans = -2e9;
    int mid = (l + r) >> 1;
    if (jobl <= mid) {
        ans = max (ans, query (jobl, jobr, l, mid, i << 1));
    }
    if (jobr > mid) {
        ans = max (ans, query (jobl, jobr, mid + 1, r, i << 1 | 1));
    }
    return ans;
}

void Update (int jobi, int jobv, int l, int r, int i) {
    if (l == r) {
        if (Max[i] < jobv) {
            Max[i] = jobv;
        } 
        return;
    }
    
    int mid = (l + r) >> 1;
    if (jobi <= mid) {
        Update (jobi, jobv, l, mid, i << 1);
    }
    if (jobi > mid) {
        Update (jobi, jobv, mid + 1, r, i << 1 | 1);
    }
    PushUp (i);   

}

int main ()
{
    ios::sync_with_stdio (false);
    cin.tie (0);

    int n, m;
    cin >> n >> m;

    arr.assign (n + 1, 0);
    Max.assign (n << 2, -1);
    for (int i = 1; i <= n; i ++)  cin >> arr[i];
    build (1, n, 1);

    for (int i = 1; i <= m; i ++) {
        char op;
        int a, b;
        cin >> op >> a >> b;

        if (op == 'Q') {
            long long ans = query (a, b, 1, n, 1);
            cout << ans << '\n';
        }
        if (op == 'U') {
            Update (a, b, 1, n, 1);
        }
    }
    return 0;
}
```
